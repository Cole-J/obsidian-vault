
[[-introduction to sql (chapter 3)]]

class participation 5
	"find the names of all instructors whose salary is greater than at least one instructor in the finance department"
	-
	attempt:
		select distinct name from instructor where salary > (something)
		(something) = (select min(salary) from instructor where dept_name="finance")
	-
	actual:
		select distinct T.name
		from instructor as T, instructor as S
		where T.salary > S.salary and S.dept_name ="finance";
