class participation 3
	member(memb_no, name, age)
	book(isbn, title, author, publisher)
	borrowed(memb_no, isbn, data)
	"find the name of members who have borrowed ay book published by "McGraw-Hill""
	in sql:
		select name from member, book, borrowed where member.memb_no = borrowed.memb_no and borrowed.isbn = book.isbn and publisher = "McGraw-Hill";
